# 🖥️ فصل هشتم: طراحی رابط کاربری (GUI) با Tkinter

## 🎯 درس ۸.۸: مدیریت چند صفحه در یک پنجره با زبانه‌ها (`CTkTabview`)

---

### 📑 تب‌ها یعنی چی؟

حالا تصور کن چت‌روم برنامه **Anonyma** باز شده؛ ما نیاز داریم کاربر بتونه هم‌زمان هم پیام‌هاش رو ببینه، هم به بخش تنظیمات کاربری سر بزنه و هم لیست افراد آنلاین رو چک کنه. اگر بخوایم همه این‌ها رو توی یک صفحه جا بدیم، برنامه‌مون بیش از حد شلوغ و سردرگم‌کننده میشه.

بهترین راهکار دسکتاپی برای این کار، استفاده از **تب‌ها یا زبانه‌ها (Tabs)** است؛ دقیقاً مثل مرورگر کروم که می‌تونی بین تب‌های مختلف جابه‌جا بشی!

در CustomTkinter یک ویجت فوق‌العاده کاربردی و مدرن به نام **`CTkTabview`** داریم که کارش ساختن تب‌های مختلف درون یک پنجره است. هر تب مثل یک بوم نقاشی مستقل عمل می‌کنه که می‌تونی ویجت‌های مخصوص به خودش رو داخلش بچینی.

---

### 💻 چطور یک برنامه چند تب بسازیم؟

بیا با هم ساختار چت‌روم، تنظیمات و لیست کاربران آنلاین برنامه Anonyma رو در قالب ۳ تب جداگانه پیاده‌سازی کنیم:

```python
import customtkinter as ctk
from tkinter import messagebox

ctk.set_appearance_mode("dark")
ctk.set_default_color_theme("blue")

app = ctk.CTk()
app.title("Anonyma Tabs")
app.geometry("500x450")

# ۱. ساخت ویجت اصلی زبانه‌ها (Tabview)
tabview = ctk.CTkTabview(app, width=460, height=380)
tabview.pack(pady=15, padx=20)

# ۲. ایجاد و اضافه کردن زبانه‌های دلخواه
tabview.add("💬 چت‌روم")
tabview.add("👥 افراد آنلاین")
tabview.add("⚙️ تنظیمات")

# ================================================
# ۳. طراحی داخل تب اول: "چت‌روم"
# ================================================
chat_tab = tabview.tab("💬 چت‌روم")

chat_label = ctk.CTkLabel(
    chat_tab, 
    text="💬 به اتاق گفت‌وگوی ناشناس خوش آمدید", 
    font=("Tahoma", 14, "bold"),
    text_color="cyan"
)
chat_label.pack(pady=10)

# چت‌باکس برای نمایش پیام‌ها
chat_box = ctk.CTkTextbox(chat_tab, width=420, height=180, font=("Tahoma", 12))
chat_box.insert("0.0", "🤖 سیستم: شما به چت‌روم متصل شدید.\n")
chat_box.insert("end", "💬 پیام‌های خود را ارسال کنید...\n")
chat_box.configure(state="disabled")  # غیرفعال برای ویرایش (فقط نمایش)
chat_box.pack(pady=5)

# فریم برای ورودی پیام
input_frame = ctk.CTkFrame(chat_tab)
input_frame.pack(pady=10, padx=10, fill="x")

message_entry = ctk.CTkEntry(
    input_frame, 
    placeholder_text="پیام خود را بنویسید...", 
    width=350,
    height=40
)
message_entry.pack(side="left", padx=5)

def send_message():
    text = message_entry.get().strip()
    if text:
        # فعال کردن موقت برای نوشتن
        chat_box.configure(state="normal")
        chat_box.insert("end", f"🙋 شما: {text}\n")
        chat_box.see("end")  # اسکرول به پایین
        chat_box.configure(state="disabled")
        message_entry.delete(0, "end")
    else:
        messagebox.showwarning("هشدار", "لطفاً پیامی بنویسید!")

def send_on_enter(event):
    send_message()

send_button = ctk.CTkButton(
    input_frame, 
    text="📤 ارسال", 
    command=send_message,
    width=80,
    height=40,
    corner_radius=10
)
send_button.pack(side="left", padx=5)

# اتصال کلید Enter به تابع ارسال
message_entry.bind("<Return>", send_on_enter)

# ================================================
# ۴. طراحی داخل تب دوم: "افراد آنلاین"
# ================================================
online_tab = tabview.tab("👥 افراد آنلاین")

online_label = ctk.CTkLabel(
    online_tab, 
    text="👥 کاربران فعال در این اتاق", 
    font=("Tahoma", 14, "bold"),
    text_color="green"
)
online_label.pack(pady=15)

# فریم برای لیست کاربران
users_frame = ctk.CTkFrame(online_tab)
users_frame.pack(pady=5, padx=20, fill="both", expand=True)

# لیست کاربران فرضی
users = [
    "🟢 کاربر ناشناس ۱ (آنلاین)",
    "🟢 کاربر ناشناس ۲ (شما)",
    "🟡 کاربر ناشناس ۳ (غایب)",
    "🔴 کاربر ناشناس ۴ (آفلاین)"
]

for user in users:
    user_lbl = ctk.CTkLabel(
        users_frame, 
        text=f"• {user}", 
        font=("Tahoma", 12),
        anchor="w"
    )
    user_lbl.pack(anchor="w", padx=20, pady=3)

# ================================================
# ۵. طراحی داخل تب سوم: "تنظیمات"
# ================================================
settings_tab = tabview.tab("⚙️ تنظیمات")

settings_label = ctk.CTkLabel(
    settings_tab, 
    text="⚙️ تنظیمات حساب کاربری", 
    font=("Tahoma", 14, "bold"),
    text_color="yellow"
)
settings_label.pack(pady=15)

# سوئیچ حالت ناشناس
anonymous_switch = ctk.CTkSwitch(
    settings_tab, 
    text="حالت ناشناس کامل (مخفی کردن IP)", 
    font=("Tahoma", 12)
)
anonymous_switch.pack(pady=10, padx=20, anchor="w")

# سوئیچ اعلان‌ها
notification_switch = ctk.CTkSwitch(
    settings_tab, 
    text="دریافت اعلان‌های جدید", 
    font=("Tahoma", 12)
)
notification_switch.select()  # روشن پیش‌فرض
notification_switch.pack(pady=10, padx=20, anchor="w")

# انتخاب تم
theme_label = ctk.CTkLabel(
    settings_tab, 
    text="انتخاب تم:", 
    font=("Tahoma", 12)
)
theme_label.pack(pady=10, padx=20, anchor="w")

theme_option = ctk.CTkOptionMenu(
    settings_tab, 
    values=["آبی", "سبز", "آبی تیره", "بنفش"],
    font=("Tahoma", 12)
)
theme_option.pack(pady=5, padx=20, anchor="w")

# دکمه ذخیره تنظیمات
def save_settings():
    messagebox.showinfo("تنظیمات", "✅ تنظیمات با موفقیت ذخیره شد!")

save_btn = ctk.CTkButton(
    settings_tab, 
    text="💾 ذخیره تنظیمات", 
    command=save_settings,
    width=200,
    height=40
)
save_btn.pack(pady=20)

app.mainloop()
```

### 🔍 ویژگی‌های کلیدی `CTkTabview`

| ویژگی | توضیح | مثال |
|-------|-------|------|
| `add("نام")` | اضافه کردن تب جدید | `tabview.add("چت‌روم")` |
| `tab("نام")` | دسترسی به یک تب خاص | `tabview.tab("چت‌روم")` |
| `select("نام")` | انتخاب یک تب به صورت پیش‌فرض | `tabview.select("چت‌روم")` |
| `delete("نام")` | حذف یک تب | `tabview.delete("تنظیمات")` |

---

### 📝 نکات مهم در طراحی تب‌ها

1. **هر تب یک والد مستقل داره:**
   ```python
   # والد هر ویجت باید tabview.tab("نام تب") باشه
   label = ctk.CTkLabel(tabview.tab("چت‌روم"), text="سلام")
   ```

2. **می‌تونی تب‌ها رو با `select()` انتخاب کنی:**
   ```python
   tabview.select("چت‌روم")  # تب چت‌روم پیش‌فرض انتخاب بشه
   ```

3. **هر تب می‌تونه هر نوع ویجتی داشته باشه:**
   - `CTkLabel`، `CTkButton`، `CTkEntry`
   - `CTkTextbox`، `CTkSwitch`، `CTkOptionMenu`
   - حتی `CTkFrame` برای گروه‌بندی

---

### 📌 اشتباهات رایج در این درس

> **🔴 فراموش کردن `tabview.tab("نام")` به عنوان والد:**
```python
# ❌ اشتباه: ویجت داخل هیچ تبی قرار نمی‌گیره
label = ctk.CTkLabel(tabview, text="سلام")

# ✅ درسته: ویجت داخل تب "چت‌روم" قرار می‌گیره
label = ctk.CTkLabel(tabview.tab("چت‌روم"), text="سلام")
```

> **🟡 قاطی کردن `add` با `tab`:**
```python
# add: برای ایجاد تب جدید
tabview.add("چت‌روم")

# tab: برای دسترسی به تب موجود
chat_tab = tabview.tab("چت‌روم")
```

> **🟢 فراموش کردن `pack()` یا `grid()` برای ویجت‌های داخل تب:**
```python
label = ctk.CTkLabel(tabview.tab("چت‌روم"), text="سلام")
# ❌ فراموش شده: label.pack()
```

> **🟣 استفاده از نام تب با فاصله و کاراکترهای خاص:**
```python
# ✅ درسته: می‌تونی از ایموجی و فاصله استفاده کنی
tabview.add("💬 چت‌روم")
chat_tab = tabview.tab("💬 چت‌روم")  # دقیقاً همون اسم
```

### 🛠️ نوبت توئه: چالش ارسال پیام در چت‌روم!

بیا تب چت‌روم این برنامه رو کاملاً زنده و تعاملی کنیم تا بشه داخلش تایپ کرد و پیام فرستاد!

**صورت مسئله:**

1. کدهای بالا رو کپی و اجرا کن تا ساختار جذاب ۳ تب رو ببینی و بینشون جابه‌جا بشی.
2. حالا تغییرات زیر رو اعمال کن تا بشه در تب اول پیام فرستاد:
   - در تب **"چت‌روم"**، زیر چت‌باکس بزرگ (`chat_box`)، یک کادر ورودی کوچک (`CTkEntry`) به نام `message_entry` بساز تا کاربر بتونه پیامش رو توش بنویسه.
   - در کنار یا زیر آن، یک دکمه (`CTkButton`) با متن `"ارسال"` اضافه کن.
   - یک تابع به نام `send_message` بساز. کار این تابع این باشه:
     1. متن داخل `message_entry` رو بخونه.
     2. اون متن رو به انتهای چت‌باکس بزرگ اضافه کنه. *(راهنمایی: برای اضافه کردن متن به چت‌باکس از این دستور استفاده کن: `chat_box.insert("end", f"شما: {text}\n")`)*.
     3. بعد از ارسال، کادر ورودی پیام (`message_entry`) رو خالی کنه.
   - دکمه ارسال رو به این تابع متصل کن.

### 💪 چالش دوم (اختیاری - برای توسعه‌دهندگان برنامه‌های چندمنظوره!)

حالا که با تب‌ها آشنا شدی، یه **برنامه مدیریت تسک (Task Manager)** با ۳ تب طراحی کن:

1. پنجره با عنوان `"📋 مدیریت تسک‌ها"` و اندازه `600x450`.

2. **تب اول: `"📝 لیست تسک‌ها"`**:
   - یک `CTkTextbox` برای نمایش لیست تسک‌ها
   - یک `CTkEntry` برای نوشتن تسک جدید
   - یک دکمه `"افزودن تسک"` که تسک رو به Textbox اضافه کنه
   - یک دکمه `"پاک کردن همه"` که با تأیید `messagebox` همه رو پاک کنه

3. **تب دوم: `"📊 آمار"`**:
   - نمایش تعداد کل تسک‌ها
   - نمایش تعداد تسک‌های انجام‌شده
   - نمایش درصد پیشرفت

4. **تب سوم: `"⚙️ تنظیمات"`**:
   - سوئیچ برای حالت تاریک/روشن (با `ctk.set_appearance_mode()`)
   - انتخاب اولویت پیش‌فرض (`CTkOptionMenu`)
   - دکمه `"ذخیره تنظیمات"` با پیام موفقیت
